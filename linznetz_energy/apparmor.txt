#include <tunables/global>

# AppArmor profile of the add-on: what the Python main loop and its page may do. Home Assistant
# loads it instead of the default profile of add-ons.
#
# Enforced: access outside these rules is denied and logged by the host (apparmor="DENIED").
# See DOCS.md, "Security".
profile linznetz_energy flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Stopped by the Supervisor (docker stop)
  signal (receive) set=(term,int,hup,kill),

  # Network: LINZ NETZ portal (HTTPS), Home Assistant websocket and Supervisor API, MQTT broker,
  # its page on port 8099 (Ingress); DNS
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,

  # Python and its libraries; no other program is started
  /usr/local/bin/python3* ix,
  /usr/local/lib/** mr,
  /usr/lib/** mr,
  /lib/** mr,
  /app/ r,
  /app/** r,

  # System files Python and the TLS connections read: certificates, DNS, time zones, users
  /etc/** r,
  /usr/share/** r,
  owner @{PROC}/@{pid}/** r,
  /proc/sys/kernel/random/** r,
  /sys/devices/system/cpu/** r,
  /dev/null rw,
  /dev/urandom r,
  /dev/random r,

  # Its own files: options.json (read), the add-on config folder with the readings, the live
  # archive, the import folder and the page status
  /data/ r,
  /data/** rw,
  /config/ r,
  /config/** rwk,
  /tmp/** rwk,
}
